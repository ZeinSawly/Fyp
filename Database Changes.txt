For the Fees in the SIS, I created the following new tables + updates student transactions table:

CREATE TABLE credit_pricing (
  id INT NOT NULL AUTO_INCREMENT,
  major_id INT NOT NULL,
  price_per_credit DECIMAL(10,2) NOT NULL,
  effective_from DATE NOT NULL,
  effective_until DATE DEFAULT NULL,  -- NULL = currently active
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  PRIMARY KEY (id),
  KEY idx_major (major_id),
  KEY idx_effective (effective_from, effective_until),
  
  CONSTRAINT fk_pricing_major FOREIGN KEY (major_id) REFERENCES majors(id) ON DELETE CASCADE
);


CREATE TABLE fee_types (
  id INT NOT NULL AUTO_INCREMENT,
  code VARCHAR(30) NOT NULL UNIQUE,
  name VARCHAR(100) NOT NULL,
  description TEXT,
  charge_basis ENUM('per_semester', 'one_time', 'on_demand') NOT NULL,
  is_active TINYINT(1) DEFAULT 1,
  
  PRIMARY KEY (id)
);


CREATE TABLE fee_pricing (
  id INT NOT NULL AUTO_INCREMENT,
  fee_type_id INT NOT NULL,
  major_id INT DEFAULT NULL,  -- NULL = applies to all majors
  amount DECIMAL(10,2) NOT NULL,
  effective_from DATE NOT NULL,
  effective_until DATE DEFAULT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  PRIMARY KEY (id),
  KEY idx_fee_type (fee_type_id),
  KEY idx_major (major_id),
  
  CONSTRAINT fk_fee_pricing_type FOREIGN KEY (fee_type_id) REFERENCES fee_types(id),
  CONSTRAINT fk_fee_pricing_major FOREIGN KEY (major_id) REFERENCES majors(id)
);

I Created the payments table that records payments the officer records:


CREATE TABLE payments (
  id INT NOT NULL AUTO_INCREMENT,
  transaction_id BIGINT NOT NULL,             -- which charge this payment is for
  student_id INT NOT NULL,                    -- denormalized for fast lookup
  amount DECIMAL(10,2) NOT NULL,              -- how much was paid
  payment_method ENUM('cash','card','check','bank_transfer','other') NOT NULL,
  reference_number VARCHAR(100) DEFAULT NULL, -- check #, txn ref, receipt #
  payment_date DATE NOT NULL,                 -- when it was paid
  notes TEXT DEFAULT NULL,
  recorded_by INT NOT NULL,                   -- which finance officer (user.id)
  status ENUM('completed','reversed') NOT NULL DEFAULT 'completed',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  PRIMARY KEY (id),
  KEY idx_transaction (transaction_id),
  KEY idx_student (student_id),
  KEY idx_payment_date (payment_date),
  KEY idx_recorded_by (recorded_by),
  
  CONSTRAINT fk_payment_transaction FOREIGN KEY (transaction_id) REFERENCES student_financial_transactions(id) ON DELETE CASCADE,
  CONSTRAINT fk_payment_student FOREIGN KEY (student_id) REFERENCES students(user_id),
  CONSTRAINT fk_payment_officer FOREIGN KEY (recorded_by) REFERENCES users(id)
);

  


For the late fee payment, I created a penalties table:

CREATE TABLE penalties (
  id INT NOT NULL AUTO_INCREMENT,
  installment_id BIGINT NOT NULL,         -- which installment this penalty is for
  student_id INT NOT NULL,                 -- denormalized for fast lookup
  week_number INT NOT NULL,                -- 1, 2, 3, ... (which week of being late)
  amount DECIMAL(10,2) NOT NULL,           -- the penalty amount ($100 typically)
  transaction_id BIGINT NOT NULL,          -- the corresponding row in student_financial_transactions
  applied_date DATE NOT NULL,              -- when the penalty was triggered
  status ENUM('active', 'waived') NOT NULL DEFAULT 'active',
  waived_by INT DEFAULT NULL,              -- officer who waived it (if applicable)
  waived_at TIMESTAMP DEFAULT NULL,
  waive_reason TEXT DEFAULT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  PRIMARY KEY (id),
  UNIQUE KEY unique_week_per_installment (installment_id, week_number),
  KEY idx_installment (installment_id),
  KEY idx_student (student_id),
  KEY idx_transaction (transaction_id),
  
  CONSTRAINT fk_penalty_installment FOREIGN KEY (installment_id) REFERENCES student_financial_transactions(id) ON DELETE CASCADE,
  CONSTRAINT fk_penalty_student FOREIGN KEY (student_id) REFERENCES students(user_id),
  CONSTRAINT fk_penalty_transaction FOREIGN KEY (transaction_id) REFERENCES student_financial_transactions(id) ON DELETE CASCADE,
  CONSTRAINT fk_penalty_waived_by FOREIGN KEY (waived_by) REFERENCES users(id)
);

Added these 3 tables for the quiz:

CREATE TABLE skill_domains (
  id INT AUTO_INCREMENT PRIMARY KEY,
  code VARCHAR(50) UNIQUE NOT NULL,
  name VARCHAR(100) NOT NULL,
  description TEXT,
  allowed_languages JSON NOT NULL,
  display_order INT DEFAULT 0,
  active TINYINT(1) DEFAULT 1
);



CREATE TABLE quiz_sessions (
  id INT AUTO_INCREMENT PRIMARY KEY,
  student_id INT NOT NULL,
  domain_code VARCHAR(50) NOT NULL,
  language VARCHAR(50) NOT NULL,
  question_count INT NOT NULL DEFAULT 15,
  questions_json JSON NOT NULL,
  status ENUM('in_progress','completed','abandoned') DEFAULT 'in_progress',
  correct_count INT DEFAULT NULL,
  reasoning_avg DECIMAL(3,2) DEFAULT NULL,
  total_score DECIMAL(5,2) DEFAULT NULL,
  score_percent DECIMAL(5,2) DEFAULT NULL,
  proficiency_level VARCHAR(20) DEFAULT NULL,
  started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  completed_at TIMESTAMP NULL,
  FOREIGN KEY (student_id) REFERENCES users(id),
  INDEX idx_student (student_id),
  INDEX idx_status (status)
);




CREATE TABLE quiz_responses (
  id INT AUTO_INCREMENT PRIMARY KEY,
  session_id INT NOT NULL,
  question_index INT NOT NULL,
  question_text TEXT NOT NULL,
  question_type ENUM('mcq','code_output','true_false') DEFAULT 'mcq',
  difficulty ENUM('easy','medium','hard') DEFAULT 'medium',
  correct_answer VARCHAR(500),
  student_answer VARCHAR(500),
  student_explanation TEXT,
  is_correct TINYINT(1),
  reasoning_score DECIMAL(3,2) DEFAULT NULL,
  reasoning_feedback TEXT,
  total_score DECIMAL(3,2) DEFAULT NULL,
  answered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (session_id) REFERENCES quiz_sessions(id) ON DELETE CASCADE,
  INDEX idx_session (session_id)
);
